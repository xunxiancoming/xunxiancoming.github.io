全链路原理：写入路径与查询路径

WHY

PRINCIPLE 01

检索质量决定回答质量，生成模型只是「嘴巴」

RAG 里最容易被低估的是切分和检索。切分决定了一个知识点会不会被拦腰砍断， 检索决定了模型到底能看见什么。模型再强，检索没捞到，也只能编。 所以下面 01 到 04 节（占了五分之四篇幅）都在处理数据，第 05 节才轮到大模型出场。

01

解析 PDF：pypdf 按页产出 Document

LOAD

第一件事是把 PDF 变成 LangChain 的 Document。 每一页一个 Document，并把页码写进 metadata——后面回答问题时 能标「根据第 3 页」，全靠这里留的字段。

01_load_pdf.py PYTHON 
"""Step 1：用 pypdf 解析 assets/report.pdf，产出带页码元数据的 Document。"""

import os
from pathlib import Path

from langchain_core.documents import Document
from pypdf import PdfReader

PROJECT_ROOT = Path(__file__).resolve().parent.parent
PDF_PATH = os.getenv("REPORT_PDF", str(PROJECT_ROOT / "assets" / "report.pdf"))

print("=" * 64)
print("  Step 1  pypdf 解析 report.pdf")
print("=" * 64)

reader = PdfReader(PDF_PATH)
print(f"\n[0] 文件: {PDF_PATH}")
print(f"    页数: {len(reader.pages)}")

documents: list[Document] = []
for i, page in enumerate(reader.pages, 1):
    text = (page.extract_text() or "").strip()
    documents.append(
        Document(
            page_content=text,
            metadata={"source": Path(PDF_PATH).name, "page": i, "chars": len(text)},
        )
    )
    print(f"    第 {i} 页: {len(text)} 字符")

total = sum(d.metadata["chars"] for d in documents)
print(f"\n[1] 合计 {len(documents)} 个 Document，{total} 字符")

print("\n[2] 第 1 页前 200 字预览:")
print("    " + documents[0].page_content[:200].replace("\n", " / "))

PRINCIPLE 01

页码元数据是研报 RAG 的刚需

研报的价值大量集中在第 1 页的评级、第 2 页的盈利预测和风险提示里。 如果切分时丢掉页码，检索结果就变成一堆没有出处的碎片，模型没法说「数据来自第几页」， 读者也无法回原文核对。一个 {"page": i} 的成本，换整条链路可追溯。

注意：pypdf 抽的是纯文本。第 3 页的三张财务报表 （资产负债表、现金流量表、利润表）会被拉平成一行行数字，表格的行列关系没有了。 这是纯文本路线的固有代价，本文接受它——因为检索目标是「哪一年净利润多少」这类数值， 拉平后数字仍然在同一个块里。表格结构有强需求的话，得换版面解析（见第 06 节坑 02）。
02

中文切分：500 字一块，重叠 100 字

SPLIT

切分用 RecursiveCharacterTextSplitter， 参数 chunk_size=500、chunk_overlap=100。 关键是分隔符列表：英文默认按空格切，中文没有空格，所以显式加上 "。" "；" "，"，让它优先在句子边界下刀。

02_split.py PYTHON 
"""Step 2：中文友好的递归切分 —— chunk_size=500，chunk_overlap=100。"""

import os
from pathlib import Path

from langchain_text_splitters import RecursiveCharacterTextSplitter
from pypdf import PdfReader

PROJECT_ROOT = Path(__file__).resolve().parent.parent
PDF_PATH = os.getenv("REPORT_PDF", str(PROJECT_ROOT / "assets" / "report.pdf"))

CHUNK_SIZE = 500
CHUNK_OVERLAP = 100
SEPARATORS = ["\n\n", "\n", "。", "；", "，", "", " "]

print("=" * 64)
print("  Step 2  RecursiveCharacterTextSplitter 中文切分")
print("=" * 64)

reader = PdfReader(PDF_PATH)
pages = [(p.extract_text() or "").strip() for p in reader.pages]

splitter = RecursiveCharacterTextSplitter(
    chunk_size=CHUNK_SIZE,
    chunk_overlap=CHUNK_OVERLAP,
    separators=SEPARATORS,
    length_function=len,
)

chunks = []
for i, text in enumerate(pages, 1):
    for j, piece in enumerate(splitter.split_text(text), 1):
        chunks.append({"text": piece, "page": i, "chunk": j})

print(f"\n[0] 参数: chunk_size={CHUNK_SIZE}, overlap={CHUNK_OVERLAP}")
print(f"    分隔符: {SEPARATORS}")
print(f"\n[1] 原始 {len(pages)} 页 → 切出 {len(chunks)} 个块")

lengths = [len(c["text"]) for c in chunks]
print(f"    长度: 最短 {min(lengths)} / 平均 {sum(lengths) // len(lengths)} / 最长 {max(lengths)}")

print("\n[2] 前 3 个块预览:")
for c in chunks[:3]:
    preview = c["text"][:60].replace("\n", " ")
    print(f"    [p{c['page']} #{c['chunk']}] {len(c['text'])} 字 | {preview}...")

PRINCIPLE 02

重叠区是为了不让关键数字死在边界上

5 页 5556 字符切出 15 个块，平均 428 字、最长 497 字。 chunk_overlap=100 让相邻两块首尾重叠约 100 字： 「2026-2028 年净利润分别为 2.45、7.30、11.66 亿元」这种长句即使正好压在切分线上， 也会在下一块里重复出现一次，不会出现「上半句在 A 块、数字在 B 块」的断裂。

参数怎么选的：500 字是中文研报的常见经验值—— 足够装下一个完整论点（含标题和支撑数据），又不至于把三个不相关话题揉进同一块。 这次的 15 个块规模下 500/100 的效果足够好，文档更长时可以考虑按小节（▌ 标题）切。
03

向量入库：embeddinggemma → Redis Stack

INDEX

入库这一步做三件事：删旧索引（保证脚本可重复跑）、建索引、写向量。 索引配置是 HNSW 图索引 + COSINE 距离， 维度 768 对齐 embeddinggemma 的输出；元数据三个字段： source（tag）、page（tag）、 chunk（numeric），前两个能做等值过滤。

03_index_redis.py PYTHON 
"""Step 3：embeddinggemma 向量化 → 写入 Redis Stack（HNSW + COSINE）。"""

import os
from pathlib import Path

from langchain_core.documents import Document
from langchain_text_splitters import RecursiveCharacterTextSplitter
from langchain_redis import RedisConfig, RedisVectorStore
from langchain_ollama import OllamaEmbeddings
from pypdf import PdfReader

PROJECT_ROOT = Path(__file__).resolve().parent.parent
PDF_PATH = os.getenv("REPORT_PDF", str(PROJECT_ROOT / "assets" / "report.pdf"))
REDIS_URL = os.getenv("REDIS_STACK_URL", "redis://localhost:26379")
EMBED_MODEL = os.getenv("EMBED_MODEL", "embeddinggemma")
INDEX_NAME = "report_rag_index"

CHUNK_SIZE = 500
CHUNK_OVERLAP = 100
SEPARATORS = ["\n\n", "\n", "。", "；", "，", "", " "]

print("=" * 64)
print("  Step 3  写入 Redis Stack 向量索引")
print("=" * 64)

# --- 载入 + 切分（与 02 完全一致的参数）---
reader = PdfReader(PDF_PATH)
splitter = RecursiveCharacterTextSplitter(
    chunk_size=CHUNK_SIZE, chunk_overlap=CHUNK_OVERLAP, separators=SEPARATORS
)
documents = []
for i, page in enumerate(reader.pages, 1):
    text = (page.extract_text() or "").strip()
    for j, piece in enumerate(splitter.split_text(text), 1):
        documents.append(
            Document(
                page_content=piece,
                metadata={"source": Path(PDF_PATH).name, "page": i, "chunk": j},
            )
        )
print(f"\n[0] 待入库块数: {len(documents)}")

# --- 幂等：先删旧索引 ---
import redis

client = redis.Redis.from_url(REDIS_URL, decode_responses=True, socket_timeout=5)
try:
    client.ft(INDEX_NAME).dropindex(delete_documents=True)
    print(f"[1] 已删除旧索引 {INDEX_NAME}")
except Exception:
    print(f"[1] 索引 {INDEX_NAME} 不存在，首次创建")

config = RedisConfig(
    index_name=INDEX_NAME,
    redis_url=REDIS_URL,
    distance_metric="COSINE",
    indexing_algorithm="HNSW",
    embedding_dimensions=768,
    metadata_schema=[
        {"name": "source", "type": "tag"},
        {"name": "page", "type": "tag"},
        {"name": "chunk", "type": "numeric"},
    ],
)
store = RedisVectorStore(embeddings=OllamaEmbeddings(model=EMBED_MODEL), config=config)

keys = store.add_documents(documents)
print(f"[2] 写入 {len(keys)} 条，索引名 {INDEX_NAME}，key 前缀 {store.key_prefix}")
print(f"    示例 key: {keys[0]}")

info = client.ft(INDEX_NAME).info()
print(f"\n[3] FT.INFO: num_docs={info['num_docs']}  "
      f"num_records={info['num_records']}  "
      f"hash_indexing_failures={info['hash_indexing_failures']}")
print(f"    幻灯片式确认: FT._LIST -> {client.execute_command('FT._LIST')}")

PRINCIPLE 03

幂等：先 dropindex 再建，脚本才能随便重跑

dropindex(delete_documents=True) 把旧索引连数据一起删掉， 第二次运行时打印「已删除旧索引」，第一次打印「首次创建」——两种都退出码 0。 不做这一步，重复跑脚本会不断往同一个索引里堆重复块， 检索结果会被同一篇文章的旧副本刷屏。最后用 FT._LIST 确认：索引里现在有 report_rag_index、ollama_rag_demo、 my_index11 三个，前两个是本文和上一篇实验留下的。

性能：15 个块、768 维，整段 load → split → embed → 写入 实测 8.9 秒， 其中大头是 embeddinggemma 逐批算向量。FT.INFO 显示 num_docs=15、hash_indexing_failures=0， 没有索引失败的文档。
04

相似度检索：三个查询 + 一次页码过滤

SEARCH

索引建好后先别急着接大模型，单独验证检索。 脚本跑了三个查询，每个取 top-3，打印距离与相似度；最后用 Tag("page") == "2" 演示元数据过滤——只允许在第 2 页里找。

04_search.py PYTHON 
"""Step 4：对 report.pdf 的向量索引做相似度检索（含页码过滤）。"""

import os

from langchain_redis import RedisConfig, RedisVectorStore
from langchain_ollama import OllamaEmbeddings
from redisvl.query.filter import Tag

REDIS_URL = os.getenv("REDIS_STACK_URL", "redis://localhost:26379")
EMBED_MODEL = os.getenv("EMBED_MODEL", "embeddinggemma")
INDEX_NAME = "report_rag_index"

print("=" * 64)
print("  Step 4  相似度检索 report_rag_index")
print("=" * 64)

config = RedisConfig(
    index_name=INDEX_NAME,
    redis_url=REDIS_URL,
    from_existing=True,
    distance_metric="COSINE",
    indexing_algorithm="HNSW",
    embedding_dimensions=768,
)
store = RedisVectorStore(embeddings=OllamaEmbeddings(model=EMBED_MODEL), config=config)

QUERIES = [
    "胜蓝股份是做什么的？现在的投资评级",
    "2026 到 2028 年净利润预测是多少",
    "液冷业务进展如何",
]

for q in QUERIES:
    print(f"\n[查询] {q}")
    for i, (doc, score) in enumerate(
        store.similarity_search_with_score(q, k=3), 1
    ):
        preview = doc.page_content[:50].replace("\n", " ")
        print(f"    {i}. 距离={score:.4f} 相似度≈{1 - score:.4f} "
              f"[p{doc.metadata['page']}#{doc.metadata['chunk']}] {preview}...")

print("\n[元数据过滤] 只在第 2 页里找「液冷」")
filt = Tag("page") == "2"
for i, (doc, score) in enumerate(
    store.similarity_search_with_score("液冷业务", k=2, filter=filt), 1
):
    preview = doc.page_content[:50].replace("\n", " ")
    print(f"    {i}. 距离={score:.4f} [p{doc.metadata['page']}] {preview}...")

PRINCIPLE 04

看相似度的相对差，别看绝对值

embeddinggemma 输出的余弦相似度大多落在 0.35 ~ 0.6 这个窄区间， 0.57 不代表「很像」，只代表「在本次候选里排第一」。 有效的判断方式是看 top-1 和 top-3 差多少：第一个查询 0.538 对 0.470， 差距明显，说明头部结果稳定；「液冷业务」三条 0.449 / 0.399 / 0.368 接近， 说明这个查询的语义在文档里本来就分散，检索置信度一般。

过滤的作用：页码过滤那次，两条结果都是 [p2]——向量库先按条件圈定候选集，再在候选集内做 KNN。 真实业务里这就是「只在指定合同/章节里找答案」的实现方式，成本只是一个 tag 条件。

05_rag_chain.py PYTHON 
"""Step 5：完整 RAG —— Redis Stack 检索 + minicpm2b:4 生成回答（LCEL）。"""

import os

from langchain_core.output_parsers import StrOutputParser
from langchain_core.prompts import ChatPromptTemplate
from langchain_ollama import ChatOllama, OllamaEmbeddings
from langchain_redis import RedisConfig, RedisVectorStore

REDIS_URL = os.getenv("REDIS_STACK_URL", "redis://localhost:26379")
EMBED_MODEL = os.getenv("EMBED_MODEL", "embeddinggemma")
CHAT_MODEL = os.getenv("CHAT_MODEL", "minicpm2b:4")
INDEX_NAME = "report_rag_index"
TOP_K = 4

print("=" * 64)
print("  Step 5  RAG 链：Redis 检索 + minicpm2b:4 生成")
print("=" * 64)

config = RedisConfig(
    index_name=INDEX_NAME,
    redis_url=REDIS_URL,
    from_existing=True,
    distance_metric="COSINE",
    indexing_algorithm="HNSW",
    embedding_dimensions=768,
)
store = RedisVectorStore(embeddings=OllamaEmbeddings(model=EMBED_MODEL), config=config)
retriever = store.as_retriever(search_kwargs={"k": TOP_K})


def format_docs(docs) -> str:
    return "\n\n".join(
        f"[第{d.metadata.get('page', '?')}页] {d.page_content}" for d in docs
    )


prompt = ChatPromptTemplate.from_template(
    "你是证券研究报告问答助手。请仅根据下方参考资料回答问题，"
    "引用数据时标明来自第几页；若资料中没有答案就直说不知道，不要编造。\n\n"
    "参考资料：\n{context}\n\n问题：{question}"
)

chain = (
    {"context": retriever | format_docs, "question": lambda x: x}
    | prompt
    | ChatOllama(model=CHAT_MODEL, temperature=0)
    | StrOutputParser()
)

print(f"\n[配置] 索引={INDEX_NAME}  top_k={TOP_K}  "
      f"嵌入={EMBED_MODEL}  生成={CHAT_MODEL}")

for question in [
    "胜蓝股份的主营业务是什么？这份研报给出的投资评级和目标判断是什么？",
    "研报预测 2026-2028 年的净利润和 EPS 分别是多少？",
    "报告里提到了哪些风险？",
]:
    print(f"\n{'-' * 60}")
    print(f"[问题] {question}")
    docs = retriever.invoke(question)
    print(f"[检索 top-{TOP_K}]")
    for i, d in enumerate(docs, 1):
        preview = d.page_content[:40].replace("\n", " ")
        print(f"    {i}. p{d.metadata['page']} 距离检索 | {preview}...")
    print("[回答]")
    answer = chain.invoke(question)
    for line in answer.splitlines():
        print(f"    {line}")
